// The generated-Go corpus runner calls dispatch.go's generated public readers.
// Manifest expectations are compared only after the reader returns a verdict.
// A refusal conforms on its verdict alone (docs/rulings.json
// refusal-verdict-only, SPEC-TABLES §6.5 and §13.10): the name the reader gives
// it is printed on a FAIL line as its diagnostic and never compared.
// `runner scorer <scorer-outcomes.tsv>` grades the shared scorer outcome cases
// (tools/hostilecorpusexec/testdata/scorer-outcomes.tsv) through the same
// expected and agrees the corpus run uses.
package main

import (
	"bufio"
	"bytes"
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"os"
	"path/filepath"
	"strings"

	"github.com/mas-bandwidth/schema/v2/tables/goruntime"
)

func main() {
	if len(os.Args) == 3 && os.Args[1] == "scorer" {
		os.Exit(scorer(os.Args[2]))
	}
	if len(os.Args) != 2 {
		fmt.Fprintln(os.Stderr, "usage: go-hostile <corpus-dir>")
		fmt.Fprintln(os.Stderr, "       go-hostile scorer <scorer-outcomes.tsv>")
		os.Exit(2)
	}
	if err := run(os.Args[1]); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func run(dir string) error {
	file, err := os.Open(filepath.Join(dir, "manifest.tsv"))
	if err != nil {
		return err
	}
	defer file.Close()
	scan := bufio.NewScanner(file)
	scan.Buffer(make([]byte, 4096), 4*1024*1024)
	total, passed, failed := 0, 0, 0
	// THE LEDGER (docs/HOSTILE-CORPUS.md): the cases dispatch answered, by
	// name, which tools/hostileledger holds to the corpus folder.
	var ran []string
	seen := map[string]bool{}
	for scan.Scan() {
		line := scan.Text()
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		row := strings.Split(line, "\t")
		if len(row) != 9 {
			return fmt.Errorf("manifest row has %d columns, want9", len(row))
		}
		if seen[row[0]] {
			return fmt.Errorf("duplicate case %s", row[0])
		}
		seen[row[0]] = true
		data, err := os.ReadFile(filepath.Join(dir, filepath.FromSlash(row[4])))
		if err != nil {
			return err
		}
		total++
		got := dispatch(row[1], row[2], row[3], data)
		ran = append(ran, row[0])
		want, err := expected(row[6], row[7], row[8])
		if err != nil {
			return err
		}
		if agrees(got, want) {
			passed++
			fmt.Printf("PASS %s\n", row[0])
		} else {
			failed++
			fmt.Printf("FAIL %s carrier=%s got=%s want=%s\n", row[0], row[1], got, want)
		}
	}
	if err := scan.Err(); err != nil {
		return err
	}
	fmt.Printf("generated go: %d cases: %d pass, %d unexplained, 0 skipped\n", total, passed, failed)
	for _, name := range ran {
		fmt.Printf("hostile-ledger ran %s\n", name)
	}
	fmt.Println("hostile-ledger end")
	if total == 0 || failed != 0 {
		return fmt.Errorf("generated Go corpus disagrees: %d failures across %d cases", failed, total)
	}
	return nil
}

// expected is the manifest's answer for a case.
func expected(verdict, reason, digest string) (string, error) {
	switch verdict {
	case "accept":
		return "accept " + digest, nil
	case "refuse":
		return "refuse " + reason, nil
	}
	return "", fmt.Errorf("invalid expected verdict %q", verdict)
}

func refused(answer string) bool {
	return answer == "refuse" || strings.HasPrefix(answer, "refuse ")
}

// agrees reports whether a reader's answer conforms to the manifest's: a
// refusal on its verdict alone (refusal-verdict-only), an accept on its digest.
// An error or a panic is never a refusal.
func agrees(got, want string) bool {
	if refused(got) && refused(want) {
		return true
	}
	return got == want
}

// scorer grades each row of the shared scorer fixture: the expected verdict,
// reference diagnostic and accepted SHA-256 make the manifest's answer; the
// worker reply makes the reader's (a refusal with the name it gives or none,
// an accept of the text body's digest, or a worker that exits). agrees must
// pass exactly the rows whose scorer exit is 0.
func scorer(path string) int {
	text, err := os.ReadFile(path)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		return 2
	}
	cases, wrong := 0, 0
	for _, line := range strings.Split(string(text), "\n") {
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		f := strings.Split(line, "\t")
		if len(f) != 7 {
			fmt.Fprintf(os.Stderr, "%s: row has %d fields, want 7: %s\n", path, len(f), line)
			return 2
		}
		cases++
		var got string
		switch reply, body := f[4], f[5]; {
		case refused(reply):
			got = reply
		case reply == "accept text":
			if body == "-" {
				body = ""
			}
			sum := sha256.Sum256([]byte(body))
			got = "accept " + hex.EncodeToString(sum[:])
		case reply == "exit":
			got = "error: the worker exited with status 3"
		default:
			fmt.Fprintf(os.Stderr, "%s: %s: unknown worker reply %s\n", path, f[0], reply)
			return 2
		}
		want, err := expected(f[1], f[2], f[3])
		if err != nil {
			fmt.Fprintf(os.Stderr, "%s: %s: %v\n", path, f[0], err)
			return 2
		}
		if pass := agrees(got, want); pass != (f[6] == "0") {
			fmt.Printf("SCORER %s: graded pass=%t, the fixture says scorer exit %s\n", f[0], pass, f[6])
			wrong++
		}
	}
	fmt.Printf("generated go scorer outcomes: %d cases, %d graded as the fixture says, %d wrong\n", cases, cases-wrong, wrong)
	if wrong != 0 || cases == 0 {
		return 1
	}
	return 0
}

// corpusReason maps the generated reader's diagnostic to the corpus's own
// reason vocabulary. The shared corpus's reasons are the Go engine's
// (internal/ci/hostilecorpus), and docs/SPEC-TABLES.md holds why a read refused
// to be per-language diagnostic detail: every condition the engine groups as
// damage is damage here too, and only the reasons the engine names keep their
// name.
func corpusReason(s string) string {
	if i := strings.Index(s, ": "); i >= 0 {
		s = s[:i]
	}
	switch s {
	case "count_over_extent_cap":
		return "count_refusal"
	case "newer_form", "no_vocabulary", "second_announcement", "vocabulary_too_large",
		"message_form_as_file", "batch_too_large", "count_refusal", "non_finite_float":
		return s
	}
	return "damaged"
}

func wireRefusal(ok bool, r *goruntime.TableReport) string {
	if r.Verdict == goruntime.TableOpenRefused {
		// an unnamed refusal is still a refusal: the name is diagnostic
		return "refuse " + corpusReason(r.Reason)
	}
	switch r.Refusal {
	case goruntime.TableRefuseOk:
		if !ok {
			return "refuse damaged"
		}
		return ""
	case goruntime.TableRefuseWireDamaged, goruntime.TableRefuseMalformed, goruntime.TableRefuseTruncated:
		return "refuse damaged"
	}
	return "refuse " + corpusReason(r.Refusal.Error())
}

// The region API requires its generated measure before allocation. A named
// storage refusal comes from that public API; structural failures stay damage.
func measureRefusal(err error) string {
	if err == goruntime.TableRefuseCountOverExtentCap {
		return "refuse count_refusal"
	}
	if err == goruntime.TableRefuseNonFiniteFloat {
		return "refuse non_finite_float"
	}
	return "refuse damaged"
}

func jsonDigest(measure func() int64, write func([]byte) int64) string {
	n := measure()
	if n < 0 {
		return "error: generated JSON measure refused"
	}
	out := make([]byte, n)
	if written := write(out); written != n {
		return fmt.Sprintf("error: generated JSON wrote %d of %d bytes", written, n)
	}
	// The corpus hashes canonical content without the writer's final newline.
	sum := sha256.Sum256(bytes.TrimRight(out, "\n"))
	return fmt.Sprintf("accept %x", sum)
}

// packetDigest answers an accepted packet with the digest of the bytes the
// port's own writer wrote back.
func packetDigest(written []byte) string {
	sum := sha256.Sum256(written)
	return fmt.Sprintf("accept %x", sum)
}
